class Imagem
{
public:
    int largura, altura;
    std::vector<Pixel> pixels;

    Imagem(int w, int h) : largura(w), altura(h) { //dflt ctro inicializando com w e h
        pixels.resize(w * h);
    }

    Pixel& operator()(int x, int y) { //acesso por (x,y)
        return pixels[y * largura + x]; //? -> pq os valores estão em posições contiguas
    } //isso aqui me retorna um objeto Pixel na posição (x,y)

    //save_ppm ()

    Pixel set_pixel(int r, int g, int b) {
        Pixel* pixel = new Pixel;
        pixel->r = r;
        pixel->g = g;
        pixel->b = b;

    }
};
